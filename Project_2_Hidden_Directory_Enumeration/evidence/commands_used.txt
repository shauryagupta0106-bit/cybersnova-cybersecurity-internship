PROJECT: Hidden Directory & File Enumeration
TARGET: http://testasp.vulnweb.com/

Environment:
whoami
pwd
cat /etc/os-release
date

Setup:
sudo apt update
sudo apt install -y gobuster dirsearch dirb seclists curl wget python3 python3-venv python3-pip xdg-utils

Target checks:
curl -sS -I --max-time 20 "$TARGET/"
curl -sS -L --max-time 20 "$TARGET/" -o results/homepage.html

Gobuster:
gobuster dir -u "$TARGET/" -w "$WORDLIST" -t 10 --timeout 10s -x asp,aspx,html,txt,bak -o results/gobuster.txt

Dirsearch:
dirsearch -u "$TARGET/" -w "$WORDLIST" --threads=10 --timeout=10 --format=plain -o results/dirsearch.txt

Dirb:
dirb "$TARGET/" "$WORDLIST" -o results/dirb.txt

Review:
cat results/gobuster.txt
cat results/dirsearch.txt
cat results/dirb.txt
cat results/discovered_directories.txt
cat results/verification.txt
